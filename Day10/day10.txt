NFS / Samba : 네트워크 파일 공유 프로토콜/서비스

NFS : 리눅스 ↔ 리눅스 서버끼리 데이터를 빠르게 실시간 공유
Samba :  리눅스 ↔ 윈도우 PC 간에 파일이나 프린터 자원을 공유

/// Samba /// 

<Server>
dnf install -y samba samba-common samba-client

mkdir /sambatest

chmod 777 /sambatest

useradd sambauser
	> passwd sambauser 는 하지 않는다. > samba 전용 비밀번호를 사용 

smbpasswd -a sambauser
	>> samba 전용 비밀번호 등록>> 윈도우에 삼바 계정 등록
	-x : 윈도우 계정 삭제
	-d : 비활성화
	-e : 활성화
	-n : 비밀번호 없는 계정으로
		>> 완전 삭제하고 싶다면 userdel , smbpasswd -x 다 적용해야함

vi /etc/samba/smb.conf
	> global 서버 전체에 적용, homes 해당 사용자 홈 디렉토리, printers 프린터 공유
	>> 아래 추가
[sambashare]
        path = /sambatest
        browseable = yes
        writable = yes
        guest ok = yes
        read only = no
        create mask = 0777
        directory mask = 0777

testparm
	> 문법 테스트

smbstatus


firewall-cmd --permanent --add-service=samba
firewall-cmd --reload
firewall-cmd --list-all

systemctl start smb nmb
systemctl enable smb nmb

/// rsys ///

<서버, 클라이언트>
dnf install -y rsyslog rsyslog-doc


<서버>

vi /etc/rsyslog.conf
	> 10 
	> 41번 라인에 추가 기입
	>> 	$ModLoad imudp
		$inputUDPServerRun 514

		$ModLoad imtcp
     		$inputTCPServerRun 514

firewall-cmd --permanent --add-port=514/udp
firewall-cmd --permanent --add-port=514/tcp

firewall-cmd --reload
firewall-cmd --list-all

systemctl start rsyslog
systemctl enable rsyslog


dnf install -y net-tools lsof

lsof -i tcp:514
netstat -natpl | grep 514
tail -f /var/log/messages
	>> 514 관련 검색 안됨

systemctl restart rsyslog
netstat -natpl | grep 514
	>> 적용하고 나니 검색 됨



<클라이언트>

	>> 추가 기입
	83 *.* @@192.168.16.95:514		>> udp

systemctl start rsyslog
systemctl enable rsyslog

	>> 클라이언트라 방화벽 작업은 생략
