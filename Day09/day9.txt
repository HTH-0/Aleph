///DNS 세팅///


named-checkconf /etc/named.conf
	>> conf 파일 문법 이상은 없는지 체크

cmd 창에서
	nslookup
	기본 서버:  kns.kornet.net
	Address:  168.126.63.1
	>>> 기본 dns 서버 찾기


vi /var/named/htht.com.zone
	>>zone 파일 만들고 수정

[작성내용]
$TTL 86400
@	IN      SOA     ns.htht.com. root.htht.com. (
                        20260824
                        24H
                        15M
                        48H
                        86400 );
	IN      NS      ns.htht.com.
	IN      A       192.168.16.47
ns	IN      A       192.168.16.47
www	IN      A       192.169.16.47

[설명]
$TTL 86400: 정보 저장(캐시) 시간
@ IN SOA ns.htht.com. root.htht.com.: 원본 설정 선언, 네임서버 및 관리자 이메일
20260824(Serial): 설정 파일 버전 번호
24H(Refresh): 보조 서버 동기화 주기
15M(Retry): 재시도 대기 시간
48H(Expire): 데이터 유효 만료 시간
86400(Minimum TTL): 존재하지 않는 도메인 응답 기억 시간
IN NS ns.htht.com.: 네임서버 이름 등록
IN A 192.168.16.47: htht.com 기본 IP
ns IN A 192.168.16.47: ns.htht.com 네임서버 IP
www IN A 192.169.16.47: [www.htht.com](https://www.htht.com) 웹서버 IP


named-checkzone htht.com /var/named/htht.com.zone
	>> zone 파일 문법 검사

firewall-cmd --permanent --add-service=dns
	>> 방화벽에 영구적으로 dns 서비스 등록 ( 포트로 하려면 tcp , udp 다 열어야함)

firewall-cmd --reload
	>> 적용

firewall-cmd --list-all
	>> 적용 된 부분 확인 ( service 쪽에서 dns 확인 )

systemctl start named
	>> named(dns) 서비스 시작한다

systemctl enable named
	>> dns 서비스가 재부팅해도 적용되게 한다
