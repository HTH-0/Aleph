/// Docker 설치 준비 ///


sudo apt update -y
	>> 시스템의 패키지 목록을 최신 상태로 새로고침

sudo apt install -y ca-certificates curl gnupg

 ls /etc/apt
	>> keyrings 디렉토리 확인

sudo install -m 0755 -d /etc/apt/keyrings

sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg

ls /etc/apt/keyrings
	> docker.gpg 있는지 확인

sudo chmod a+r /etc/apt/keyrings/docker.gpg

echo "deb [arch="$(dpkg --print-architecture)" signed-by=/etc/apt/keyrings/docker.gpg] https://download.docker.com/linux/ubuntu       "$(. /etc/os-release && echo "$VERSION_CODENAME")" stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

	>>  cat /etc/apt/sources.list.d/docker.list 여기에 잘 나오는지 확인한다

sudo apt update -y
	>> 저장소를 변경했으면 갱신해준다

sudo apt install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

sudo apt install -y mariadb-server

sudo systemctl start mariadb
sudo systemctl enable mariadb
sudo mysql_secure_installation
	>> 보안설치

// Maria DB 들어가서 //

create user 'pmm'@'localhost' identified by 'pmm1234' with max_user_connections 10;
	>> PMM 전용 계정 생성, 제한 설정

grant select, process, replication client, reload on *.* to 'pmm'@'localhost';
	>> 계정에 권한(조회 관련) 부여
	
flush privileges;

// 리눅스 //

sudo pmm-admin add mysql --username=pmm --password=pmm1234 --host=localhost --port=3306 --query-source=slowlog --environment=production MySQL-Primary
	>> mariadb에서 만든 계정을 pmm-admin 을 통해서 mysql 모니터링 서비스 추가

sudo pmm-admin status
	>> 확인해보기
