ALTER

alter table student add address varchar(8);
	>> student 테이블을 수정한다, 테이블에 address 라는 컬럼을 추가한다

alter table student add age int(3) after date;
	>> 테이블을 추가할때 after date >> date 컬럼 뒤에 추가한다 / 젤 위로 오게 하고싶다면 first

alter table student add (age2 int(3) age3 int(3));
	>> 여러가지 컬럼을 추가할때는 () 안에 집어넣고 ,로 구분한다.

alter table student modify name varchar(8) first;
	>> 컬럼을 수정한다. name 컬럼이 처음에 오도록

alter table student modify age int(3) after address;
	>> 컬럼을 수정한다. age 컬럼이 address 컬럼 뒤에 오도록

 alter table student drop age2;
	>> 컬럼을 삭제한다

alter table student change mbti subject varchar(4);
	>> 컬럼 이름 수정



INSERT

insert into student values ( '데이터', '데이터', 숫자데이터 ... );
	>> student 테이블에 값을 집어넣는다
	>> ( )안에 컬럼을 포함하고, 문자는 ' ' 안에 데이터를 집어넣고 숫자는 그냥 넣기

insert into student (컬럼명1, 컬럼명2 ) values( '데이터1', 숫자데이터);


UPDATE

update student set date = 2026;
	>> student 테이블의 date 컬럼의 모든 내용을 2026으로 갱신한다.

update student set address = 'daegu' where name = 'kim';
	>> student 테이블의 name 값이 'kim'인 행(Row)을 찾아 address 컬럼값을 'daegu'로 갱신한다.

update student set name = 'park', date = 2020 where name = 'kim';
	>>  여러개의 컬럼을 수정하고 싶다면 , 로 구분해서 적는다.


DELETE

delete from student where name = 'park';
	>> name 컬럼이 park 인 행을 삭제한다.

 delete from student;
	>> student 안의 모든 데이터삭제

	>> 테이블 삭제

/etc/httpd/conf/ssl << 여기에 담는다
mkdir /etc/httpd/conf/ssl
cd /etc/httpd/conf/ssl

openssl genrsa -des3 -out test.com.key
	openssl genrsa: OpenSSL을 사용하여 RSA 개인키를 생성
	-des3: 생성되는 개인키를 Triple-DES(3DES) 알고리즘으로 암호화
	-out test.com.key: 생성된 키를 저장할 파일명


openssl req -newkey rsa:2048 -nodes -keyout /etc/httpd/conf/ssl/test.com.key -x509 -days 365 -out /etc/httpd/conf/ssl/test.com.crt
	openssl req : OpenSSL 프로그램에 인증서 요청(CSR) 관리 및 X.509 인증서 생성 모드
	-x509 : 자기 자신이 서명한(Self-Signed) 인증서를 직접 출력
	-newkey rsa:2048 : 방식을 RSA, 길이를 표준 안전 규격인 2048비트 의 새로운 개인키
	-nodes :개인키에 비밀번호(Passphrase)를 걸지 않음
	-keyout /etc/httpd/conf/ssl/test.com.key : 생성된 개인키(Private Key)를 지정한 경로와 파일명으로 저장
	-out /etc/httpd/conf/ssl/test.com.crt : 생성된 인증서(Certificate)를 지정한 경로와 파일명으로 저장
	-days 365 : 유효기간 1년

ftp / ftps / sftp

FTP 포트 구성
21번 포트: 로그인, 파일 목록 조회, 삭제 등 명령어를 주고받는 제어 채널.
20번 포트: Active 모드에서 실제 파일 데이터를 전송하는 데이터 채널
1024번 이상 임의 포트: Passive 모드에서 서버가 파일 데이터 전송을 위해 동적으로 개방하여 사용하는 데이터 채널

FTP 연결 방식
Active 모드: 클라이언트가 21번 포트로 연결을 요청하면, 서버가 20번 포트를 통해 클라이언트 측으로 역방향 데이터 연결을 시도하는 방식
Passive 모드: 클라이언트가 21번 포트로 접속한 뒤, 서버가 안내해 주는 1024번 이상의 임의 포트로 클라이언트가 직접 순방향 데이터 연결을 시도하는 방식
